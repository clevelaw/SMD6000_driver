    def send_command(self, cmd):
        self.ser.reset_input_buffer()
        print(f"Sending: {cmd}")

        self.ser.write((cmd + '\r').encode())
        # Wait for PSU to send everything
        time.sleep(3)

        # Read all available bytes at once
        waiting = self.ser.in_waiting
        print(f"Bytes in buffer: {waiting}")
        raw_all = self.ser.read(waiting) if waiting else b''
        print(f"All raw bytes: {raw_all}")
        print(f"All raw hex: {raw_all.hex()}")

        # Split into lines and print each one
        lines = raw_all.split(b'\r')
        for i, line in enumerate(lines):
            print(f"Line {i}: {repr(line)}")

        return raw_all.decode('ascii', errors='replace').strip()
